Saga Pattern (Distributed Transactions)

flowchart LR
    subgraph Orchestrator["Saga Orchestrator"]
        SO[Saga Coordinator]
    end
    
    subgraph Steps["Saga Steps"]
        S1[1. Create Order]
        S2[2. Reserve Stock]
        S3[3. Process Payment]
        S4[4. Ship Order]
    end
    
    subgraph Compensations["Compensation Actions"]
        C1[Cancel Order]
        C2[Release Stock]
        C3[Refund Payment]
        C4[Cancel Shipment]
    end
    
    SO -->|execute| S1
    S1 -->|success| S2
    S2 -->|success| S3
    S3 -->|success| S4
    S4 -->|complete| SO
    
    S4 -.->|failure| C4
    C4 -.-> C3
    C3 -.-> C2
    C2 -.-> C1
    C1 -.->|rollback complete| SO
